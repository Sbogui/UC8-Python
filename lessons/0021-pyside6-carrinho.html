Antes da aula — Teoria

Conceito

Qt Widgets organiza controles por layout. Um sinal descreve um evento e pode ser ligado a uma função para produzir um efeito na tela.

Exemplo mínimo comentado

Uma janela que responde a um botão

from PySide6.QtWidgets import QApplication, QLabel, QPushButton, QVBoxLayout, QWidget

app = QApplication([])
janela = QWidget()
janela.setWindowTitle("Ensaio de sinais")
layout = QVBoxLayout(janela)
mensagem = QLabel("Aguardando o clique")
botao = QPushButton("Testar sinal")
layout.addWidget(mensagem)
layout.addWidget(botao)
# Passe a função ao connect para o Qt chamá-la no clique.
botao.clicked.connect(lambda: mensagem.setText("Clique recebido"))
janela.show()
app.exec()

Execute ensaio_sinais.py em uma máquina com PySide6 e uma sessão gráfica.

Resultado esperado do ensaio
A janela abre com Aguardando o clique. Ao acionar Testar sinal, o rótulo muda para Clique recebido.

Três erros comuns

  1. Executar com um Python diferente do ambiente em que PySide6 foi instalado.
  2. Chamar a função no momento de conectar em vez de passar a função.
  3. Associar o produto à posição visual da linha.

Monte o primeiro fluxo com Qt Widgets ↗

1. Problema concreto

Como fazer um botão disparar uma ação em uma interface controlada por eventos?

2. Entenda a ideia

PySide6 liga Python ao Qt 6. QApplication controla a aplicação. Um widget representa uma parte da tela. O layout distribui os componentes quando a janela muda de tamanho.

O sinal clicked é emitido quando o botão é acionado. connect liga o sinal a uma função. No carrinho, a janela lê o campo e a seleção, o modelo aplica a regra e a janela comunica o resultado.

3. Exemplo comentado

Preparação no PowerShell
py -m venv .venv
.\.venv\Scripts\python.exe -m pip install PySide6
.\.venv\Scripts\python.exe -c "import PySide6; print(PySide6.__version__)"
.\.venv\Scripts\python.exe ensaio_sinais.py

Execute os comandos na pasta da A21. O último comando abre a janela do anexo ensaio_sinais.py.

Desafio · edite o carrinho com PySide6

Coloque o arquivo inicial, carrinho.qss e seu modelos.py funcional na mesma pasta. Implemente atualizar_quantidade. A tabela e o total devem refletir a regra já desenvolvida no projeto.

  2. Localize selectionModel().selectedRows() e recupere o ID da linha selecionada.
  3. Passe o ID e o texto do campo para Pedido.atualizar. Use a mensagem do domínio quando uma edição for recusada e deixe o campo pronto para correção.
  4. Atualize a tabela e o total depois de uma edição aceita. Em carrinho.qss, ajuste contraste, espaçamento ou destaque do total e confira o foco com Tab.

from pathlib import Path

from PySide6.QtCore import Qt
from PySide6.QtWidgets import (
    QAbstractItemView, QApplication, QHeaderView, QHBoxLayout, QLabel,
    QLineEdit, QPushButton, QTableWidget, QTableWidgetItem, QVBoxLayout, QWidget,
)

from modelos import Pedido, Produto, formatar


def moeda(centavos):
    return f"R$ {formatar(centavos)}"


def pedido_de_ensaio():
    pedido = Pedido()
    pedido.adicionar(Produto(1, "Picolé de fruta", 500), "2")
    pedido.adicionar(Produto(2, "Picolé de leite", 750), "3")
    return pedido


class JanelaCarrinho(QWidget):
    def __init__(self, pedido):
        super().__init__()
        self.pedido = pedido
        self.setObjectName("carrinho")
        self.setWindowTitle("Carrinho de vendas")
        self.resize(740, 480)

        layout = QVBoxLayout(self)
        layout.setContentsMargins(24, 24, 24, 24)
        layout.setSpacing(14)

        titulo = QLabel("Carrinho de vendas")
        titulo.setObjectName("titulo")
        apoio = QLabel("Selecione um produto para editar sua quantidade.")
        apoio.setObjectName("apoio")
        layout.addWidget(titulo)
        layout.addWidget(apoio)

        self.tabela = QTableWidget(0, 4)
        self.tabela.setAccessibleName("Itens do carrinho")
        self.tabela.setHorizontalHeaderLabels(
            ["Produto", "Qtd.", "Preço unitário", "Subtotal"]
        )
        self.tabela.setEditTriggers(QAbstractItemView.EditTrigger.NoEditTriggers)
        self.tabela.setSelectionBehavior(QAbstractItemView.SelectionBehavior.SelectRows)
        self.tabela.setSelectionMode(QAbstractItemView.SelectionMode.SingleSelection)
        self.tabela.verticalHeader().hide()
        cabecalho = self.tabela.horizontalHeader()
        cabecalho.setSectionResizeMode(QHeaderView.ResizeMode.ResizeToContents)
        cabecalho.setSectionResizeMode(0, QHeaderView.ResizeMode.Stretch)
        layout.addWidget(self.tabela, 1)

        painel = QWidget()
        painel.setObjectName("edicao")
        edicao = QVBoxLayout(painel)
        edicao.setContentsMargins(14, 12, 14, 12)
        linha = QHBoxLayout()
        rotulo = QLabel("&Nova quantidade")
        self.campo = QLineEdit()
        self.campo.setAccessibleName("Nova quantidade")
        self.campo.setMaxLength(12)
        rotulo.setBuddy(self.campo)
        self.botao = QPushButton("&Atualizar quantidade")
        linha.addWidget(rotulo)
        linha.addWidget(self.campo, 1)
        linha.addWidget(self.botao)
        edicao.addLayout(linha)

        self.retorno = QLabel("Selecione uma linha para começar.")
        self.retorno.setObjectName("retorno")
        self.retorno.setWordWrap(True)
        edicao.addWidget(self.retorno)
        layout.addWidget(painel)

        self.total = QLabel()
        self.total.setObjectName("total")
        self.total.setAlignment(Qt.AlignmentFlag.AlignRight)
        layout.addWidget(self.total)

        self.setTabOrder(self.tabela, self.campo)
        self.setTabOrder(self.campo, self.botao)
        self.botao.clicked.connect(self.atualizar_quantidade)
        self.campo.returnPressed.connect(self.atualizar_quantidade)
        self.renderizar()

    def renderizar(self):
        linhas = self.pedido.linhas()
        self.tabela.setRowCount(len(linhas))
        for indice, linha in enumerate(linhas):
            valores = (
                linha["nome"],
                str(linha["quantidade"]),
                moeda(linha["preco_centavos"]),
                moeda(linha["subtotal_centavos"]),
            )
            for coluna, valor in enumerate(valores):
                celula = QTableWidgetItem(valor)
                if coluna == 0:
                    celula.setData(Qt.ItemDataRole.UserRole, linha["produto_id"])
                else:
                    celula.setTextAlignment(
                        Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignVCenter
                    )
                self.tabela.setItem(indice, coluna, celula)
        self.total.setText(f"Total do carrinho · {moeda(self.pedido.total_centavos())}")

    def mostrar_retorno(self, mensagem, erro=False):
        self.retorno.setProperty("erro", erro)
        self.retorno.setText(mensagem)
        estilo = self.retorno.style()
        estilo.unpolish(self.retorno)
        estilo.polish(self.retorno)
